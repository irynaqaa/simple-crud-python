from flask import Blueprint, request, render_template, redirect, url_for, flash from .models import User from . import db add_user_bp = Blueprint('add_user', __name__) @add_user_bp.route('/add', methods=['GET', 'POST']) def add_user(): if request.method == 'POST': name = request.form.get('name') age = request.form.get('age') email = request.form.get('email') if not name or not age or not email: flash('All fields are required!', 'danger') return render_template('add.html', error_message='All fields are required.') try: age = int(age) new_user = User(name=name, age=age, email=email) db.session.add(new_user) db.session.commit() flash('User added successfully!', 'success') return redirect(url_for('home')) except Exception as e: db.session.rollback() flash('An error occurred while adding the user.', 'danger') return render_template('add.html', error_message='An error occurred while adding the user.') return render_template('add.html')
